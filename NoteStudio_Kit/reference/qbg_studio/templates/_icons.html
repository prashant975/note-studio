{#- Inline SVG icons. No icon font, no library, no network request - the app is local and must work behind a proxy with nothing installed. Usage: {% import "_icons.html" as ico with context %} {{ ico.i('play') }} 16px, inherits colour {{ ico.i('alert-triangle', 20) }} Every glyph is a 24x24 stroked path on currentColor, so an icon always matches the text beside it. An icon never carries meaning alone: it sits next to a word, or the button gets an aria-label. -#} {%- macro i(name, size=16, cls='ico') -%} {%- endmacro -%} {#- The icon that matches a semantic state. Used by alerts and status cards so a state is never carried by colour alone. -#} {%- macro state(kind, size=16, cls='ico') -%} {%- if kind == 'ok' -%}{{ i('check-circle', size, cls) }} {%- elif kind == 'warn' -%}{{ i('alert-triangle', size, cls) }} {%- elif kind in ('bad', 'danger') -%}{{ i('alert-circle', size, cls) }} {%- else -%}{{ i('info', size, cls) }} {%- endif -%} {%- endmacro -%}
